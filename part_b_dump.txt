#define CRT_SECURE_NO_WARNINGS
#include <stdio.h>
#include <string.h>

#include <unistd.h>    // for fork
#include <sys/types.h> // for pid_t
#include <sys/wait.h>

#define STR_SEP " \t\n"

int main()
{
    char line[2048];

    while (1)
    {
        printf("$$"); // print $$ before each line
        fflush(stdout);

        if (fgets(line, 2048, stdin) == NULL)
        {
            printf("\n"); // read input from user
            return 0;     // eof
        }

        char *args[200]; // list of arguments
        args[0] = strtok(line, STR_SEP);

        int i = 0; // num of arguments
        while (args[i] != NULL)
        {
            ++i;
            args[i] = strtok(NULL, STR_SEP);
        }

        if (i == 0) // if command is empty
        {
            continue;
        }

        // check for backgroung execution and if llast token is '%', run
        // command in the backgorund
        int background = 0;

        if (strcmp(args[i - 1], "%") == 0)
        { // last character is '%'
            background = 1;
            args[i - 1] = NULL; // remove '%' in the end
            i = i - 1;

            if (i == 0) // no command, '%' is the only token
            {
                continue;
            }
        }

        // creating a child process
        pid_t pid = fork();

        if (pid < 0)
        {
            perror("fork"); // fork failed
            continue;
        }

        if (pid == 0)
        { // child process, replace child with requested program
            execvp(args[0], args);
            perror("execvp"); // command failed
            _exit(1);
        }
        else // parent process
        {
            if (!background)
            {
                if (waitpid(pid, NULL, 0) < 0)
                { // wait until child finishes
                    perror("waitpid");
                }
            }

            // in background processes reap children tht teminated, so we will have no zombies
            // -1 to check all children
            // null - no need for exit
            // WNOHANG - do not block

            while (1)
            {
                pid_t done = waitpid(-1, NULL, WNOHANG);
                if (done > 0)
                {
                    printf("%d\n", (int)done);
                    fflush(stdout);
                }
                else
                {
                    break; // no more backgrounf finished processes
                }
            }
        }
    }
}

        pid_t pid;

        while ((pid = waitpid(-1, &status, WNOHANG)) > 0) {
            printf("%d\n", (int)pid);
        }

        write(1, "$$ ", 3);

        if (fgets(line, sizeof(line), stdin) == NULL) {
            write(1, "\n", 1);
            while ((pid = waitpid(-1, &status, WNOHANG)) > 0) {
                printf("%d\n", (int)pid);
            }
            return 0;
        }

        args[0] = strtok(line, STR_SEP);
        int i = 0;
        while (args[i] != NULL) {
            ++i;
            args[i] = strtok(NULL, STR_SEP);
        }

        if (args[0] == NULL) {
            continue;
        }

        int background = 0;
        if (i > 0 && args[i - 1] != NULL &&
            args[i - 1][0] == '%' && args[i - 1][1] == '\0') {

            background = 1;
            args[i - 1] = NULL;
            if (args[0] == NULL) {
                continue;
            }
        }

        pid = fork();
        if (pid < 0) {
            perror("fork");
            continue;
        }

        if (pid == 0) {
            execvp(args[0], args);
            perror("execvp");
            _exit(1);
        }

        if (!background) {
            while (waitpid(pid, &status, 0) == -1) {
                if (errno != EINTR) {
                    perror("waitpid");
                    break;
                }
            }
        }
    }
}
